	*statsOpen
		name:Stats
		desc:Open the current production report.
		class:menuButton
		tag:utilityControl
		shown
		on click:
			if (statsPanelActive=1)
				lose statsPanelActive
				hide tag:statsContent
				show tag:productionContent
			else
				lose optionsPanelActive
				hide tag:optionsPanel
				yield 1 statsPanelActive
				hide tag:productionContent
				show tag:statsContent
			end
		end

	*optionsOpen
		name:Options
		desc:Open the in-game options.
		class:menuButton
		tag:utilityControl
		shown
		on click:
			if (optionsPanelActive=1)
				lose optionsPanelActive
				hide tag:optionsPanel
				show tag:productionContent
			else
				lose statsPanelActive
				hide tag:statsContent
				yield 1 optionsPanelActive
				hide tag:productionContent
				show tag:optionsPanel
				if (memoBoardEnabled=1)
					show memoBoardOn
					hide memoBoardOff
				else
					hide memoBoardOn
					show memoBoardOff
				end
			end
		end
	*memoBoardOn
		name:Memo Board: On
		desc:Hide the memo board.
		class:optionButton
		hidden
		tag:optionsPanel
		on click:
			if (memoBoardEnabled=1)
				lose memoBoardEnabled
				hide tag:memoBoard
				hide memoBoardOn
				show memoBoardOff
			else
				yield 1 memoBoardEnabled
				show memo1
				hide memoBoardOff
				show memoBoardOn
			end
		end
	*memoBoardOff
		name:Memo Board: Off
		desc:Show the memo board again.
		class:optionButton
		hidden
		tag:optionsPanel
		on click:
			yield 1 memoBoardEnabled
			show memo1
			hide memoBoardOff
			show memoBoardOn
		end
